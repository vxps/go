package ledger

import (
        "encoding/json"
        "fmt"
        "io"
        "maps"
)

type Budget struct {
        Category string `json:"category"`
        Limit    int    `json:"limit"`
        Period   string `json:"period,omitempty"`
}

var budgets = make(map[string]Budget)

func GetBudgets() map[string]Budget {
        return maps.Clone(budgets)
}

func SetBudget(b Budget) error {
        if err := CheckValid(&b); err != nil {
                return fmt.Errorf("invalid budget: %w", err)
        }
        budgets[b.Category] = b
        return nil
}

func LoadBudgets(r io.Reader) error {
        var loadedBudgets []Budget

        decoder := json.NewDecoder(r)

        if err := decoder.Decode(&loadedBudgets); err != nil {
                return fmt.Errorf("failed to decode budgets: %w", err)
        }

        for _, budget := range loadedBudgets {
                if err := SetBudget(budget); err != nil {
                        return fmt.Errorf(
                                "failed to load budget for category %q: %w",
                                budget.Category,
                                err,
                        )
                }
        }

        return nil
}

package ledger

import (
        "errors"
        "fmt"
        "slices"
)

var ErrBudgetExceeded = errors.New("budget exceeded")

type Transaction struct {
        ID          int
        Amount      int
        Category    string
        Description string
        Date        string
}

var transactions = make([]Transaction, 0)

func AddTransaction(tx Transaction) (Transaction, error) {
        err := CheckValid(&tx)
        if err != nil {
                return Transaction{}, fmt.Errorf("invalid transaction: %w", err)
        }

        if budget, ok := budgets[tx.Category]; ok {
                currentSpent := sumByCategory(tx.Category)

                if currentSpent+tx.Amount > budget.Limit {
                        return Transaction{}, ErrBudgetExceeded
                }
        }

        if tx.ID == 0 {
                tx.ID = len(transactions) + 1
        }

        transactions = append(transactions, tx)

        return tx, nil
}

func ListTransactions() []Transaction {
        return slices.Clone(transactions)
}

func sumByCategory(category string) int {
        var total int

        for _, tx := range transactions {
                if tx.Category == category {
                        total += tx.Amount
                }
        }

        return total
}

func Reset() {
        budgets = make(map[string]Budget)
        transactions = make([]Transaction, 0)
}

package ledger

import (
        "errors"
        "time"
)

var ErrAmountLessZero = errors.New("amount must be >0")
var ErrCategoryEmpty = errors.New("category must not be empty")
var ErrDateInvalid = errors.New("date must be in YYYY-MM-DD format")
var ErrLimitNegative = errors.New("limit must be a positive value")

type Validatable interface {
        Validate() error
}

func (t *Transaction) Validate() error {
        if t.Amount <= 0 {
                return ErrAmountLessZero
        }
        if t.Category == "" {
                return ErrCategoryEmpty
        }
        _, err := time.Parse("2006-01-02", t.Date)
        if err != nil {
                return ErrDateInvalid
        }

        return nil
}

func (b *Budget) Validate() error {
        if b.Limit <= 0 {
                return ErrLimitNegative
        }
        if b.Category == "" {
                return ErrCategoryEmpty
        }

        return nil
}

func CheckValid(v Validatable) error {
        return v.Validate()
}
